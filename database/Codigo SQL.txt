#Crear base de datos.

CREATE DATABASE IF NOT EXISTS sistema_raee;

USE sistema_raee;

#Tabla de Roles

CREATE TABLE roles (
    id_rol INT AUTO_INCREMENT PRIMARY KEY,
    nombre_rol VARCHAR(50) NOT NULL
);

#Tabla de usurios

CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL,
    correo VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    telefono VARCHAR(20),
    estado ENUM('Activo', 'Inactivo') DEFAULT 'Activo',
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    id_rol INT
);

#Tabla de empresas

CREATE TABLE empresas (
    id_empresa INT AUTO_INCREMENT PRIMARY KEY,
    nombre_empresa VARCHAR(150) NOT NULL,
    nit VARCHAR(30) UNIQUE,
    direccion VARCHAR(150),
    ciudad VARCHAR(100),
    departamento VARCHAR(100),
    telefono VARCHAR(20),
    correo VARCHAR(100)
);

#Tabla de equipos tecnologicos

CREATE TABLE equipos (
    id_equipo INT AUTO_INCREMENT PRIMARY KEY,
    codigo_equipo VARCHAR(50) UNIQUE NOT NULL,
    nombre_equipo VARCHAR(100),
    tipo_equipo VARCHAR(100),
    marca VARCHAR(100),
    modelo VARCHAR(100),
    serial_equipo VARCHAR(100),
    fecha_compra DATE,
    estado ENUM(
        'Nuevo',
        'En uso',
        'Mantenimiento',
        'Obsoleto',
        'RAEE generado'
    ) DEFAULT 'Nuevo',
    id_empresa INT,
    responsable VARCHAR(100),
    fecha_registro TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

#HISTORIAL DE TRAZABILIDAD

CREATE TABLE historial_equipos (
    id_historial INT AUTO_INCREMENT PRIMARY KEY,
    id_equipo INT NOT NULL,
    estado_anterior VARCHAR(50),
    estado_nuevo VARCHAR(50),
    observacion TEXT,
    usuario_responsable VARCHAR(100),
    fecha_cambio TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

#TABLA RESIDUOS RAEE

CREATE TABLE residuos_raee (
    id_raee INT AUTO_INCREMENT PRIMARY KEY,
    id_equipo INT NOT NULL,
    categoria VARCHAR(100),
    peso DECIMAL(10,2),
    fecha_generacion DATE,
    observaciones TEXT
);

#DISPOICION FINAL

CREATE TABLE disposicion_final (
    id_disposicion INT AUTO_INCREMENT PRIMARY KEY,
    id_raee INT NOT NULL,
    tipo_disposicion ENUM(
        'Reciclaje',
        'Reutilización',
        'Donación',
        'Destrucción certificada'
    ),
    empresa_gestora VARCHAR(150),
    certificado_disposicion VARCHAR(255),
    fecha_disposicion DATE
);

#VISTA RAEE

CREATE VIEW vista_raee AS
SELECT
e.codigo_equipo,
e.nombre_equipo,
r.categoria,
r.fecha_generacion,
d.tipo_disposicion
FROM residuos_raee r
INNER JOIN equipos e
ON r.id_equipo = e.id_equipo
LEFT JOIN disposicion_final d
ON r.id_raee = d.id_raee;

#TRAZABILIDAD AUTOMATICA

DELIMITER $$

CREATE TRIGGER trg_historial_estado
AFTER UPDATE ON equipos
FOR EACH ROW
BEGIN
    IF OLD.estado <> NEW.estado THEN
        INSERT INTO historial_equipos(
            id_equipo,
            estado_anterior,
            estado_nuevo,
            observacion,
            usuario_responsable
        )
        VALUES(
            NEW.id_equipo,
            OLD.estado,
            NEW.estado,
            'Cambio automático de estado',
            'Sistema'
        );
    END IF;
END$$

DELIMITER ;
